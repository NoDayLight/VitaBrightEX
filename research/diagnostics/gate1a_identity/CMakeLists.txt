cmake_minimum_required(VERSION 3.10)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(DEFINED ENV{VITASDK})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain file")
  else()
    message(FATAL_ERROR "Please define VITASDK")
  endif()
endif()
project(vbe_gate1a_identity C)
include("${VITASDK}/share/vita.cmake" REQUIRED)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -Werror -O2 -std=gnu99")
add_executable(vbe_gate1a_identity_kernel main.c)
target_link_libraries(vbe_gate1a_identity_kernel taihenForKernel_stub SceModulemgrForDriver_stub SceSysmemForDriver_stub SceThreadmgrForDriver_stub gcc)
set_target_properties(vbe_gate1a_identity_kernel PROPERTIES LINK_FLAGS "-nostdlib" COMPILE_FLAGS "-D__VITA_KERNEL__ -fno-pic -fno-PIC")
vita_create_self(vbe_gate1a_identity.skprx vbe_gate1a_identity_kernel CONFIG ${CMAKE_CURRENT_SOURCE_DIR}/module.yml UNSAFE)
vita_create_stubs(vbe_gate1a_identity_stubs vbe_gate1a_identity_kernel ${CMAKE_CURRENT_SOURCE_DIR}/module.yml KERNEL)
add_executable(vbe_gate1a_drain drain.c)
add_dependencies(vbe_gate1a_drain vbe_gate1a_identity_stubs)
target_link_libraries(vbe_gate1a_drain ${CMAKE_CURRENT_BINARY_DIR}/libVbeGate1Identity_stub.a SceIofilemgr_stub SceLibKernel_stub SceKernelThreadMgr_stub)
add_executable(vbe_gate1a_status status.c)
add_dependencies(vbe_gate1a_status vbe_gate1a_identity_stubs)
target_link_libraries(vbe_gate1a_status ${CMAKE_CURRENT_BINARY_DIR}/libVbeGate1Identity_stub.a SceIofilemgr_stub SceLibKernel_stub)
set(VITA_ELF_CREATE_FLAGS "")
set(VITA_MAKE_FSELF_FLAGS "")
vita_create_self(vbe_gate1a_drain.self vbe_gate1a_drain UNSAFE)
vita_create_vpk(vbe_gate1a_drain.vpk VBEG1DRN vbe_gate1a_drain.self VERSION 01.00 NAME "VBE Gate-1A Identity Drain")
vita_create_self(vbe_gate1a_status.self vbe_gate1a_status UNSAFE)
vita_create_vpk(vbe_gate1a_status.vpk VBEG1STS vbe_gate1a_status.self VERSION 01.00 NAME "VBE Gate-1A Identity Status")

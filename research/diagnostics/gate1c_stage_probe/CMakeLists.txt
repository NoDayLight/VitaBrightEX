cmake_minimum_required(VERSION 3.10)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(DEFINED ENV{VITASDK})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain file")
  else()
    message(FATAL_ERROR "Please define VITASDK")
  endif()
endif()
project(vbe_gate1c_stage_probe C)
include("${VITASDK}/share/vita.cmake" REQUIRED)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -Werror -O2 -std=gnu99")
add_executable(vbe_gate1c_stage_probe_kernel main.c)
target_link_libraries(vbe_gate1c_stage_probe_kernel taihenForKernel_stub SceModulemgrForDriver_stub SceSysmemForDriver_stub SceThreadmgrForDriver_stub gcc)
set_target_properties(vbe_gate1c_stage_probe_kernel PROPERTIES LINK_FLAGS "-nostdlib" COMPILE_FLAGS "-D__VITA_KERNEL__ -fno-pic -fno-PIC")
vita_create_self(vbe_gate1c_stage_probe.skprx vbe_gate1c_stage_probe_kernel CONFIG ${CMAKE_CURRENT_SOURCE_DIR}/module.yml UNSAFE)
vita_create_stubs(vbe_gate1c_stage_probe_stubs vbe_gate1c_stage_probe_kernel ${CMAKE_CURRENT_SOURCE_DIR}/module.yml KERNEL)
add_executable(vbe_gate1c_drain drain.c)
add_dependencies(vbe_gate1c_drain vbe_gate1c_stage_probe_stubs)
target_link_libraries(vbe_gate1c_drain ${CMAKE_CURRENT_BINARY_DIR}/libVbeGate1cStageProbe_stub.a SceIofilemgr_stub SceLibKernel_stub SceKernelThreadMgr_stub)
add_executable(vbe_gate1c_status status.c)
add_dependencies(vbe_gate1c_status vbe_gate1c_stage_probe_stubs)
target_link_libraries(vbe_gate1c_status ${CMAKE_CURRENT_BINARY_DIR}/libVbeGate1cStageProbe_stub.a SceIofilemgr_stub SceLibKernel_stub)
set(VITA_ELF_CREATE_FLAGS "")
set(VITA_MAKE_FSELF_FLAGS "")
vita_create_self(vbe_gate1c_drain.self vbe_gate1c_drain UNSAFE)
vita_create_vpk(vbe_gate1c_drain.vpk VBEG1CDR0 vbe_gate1c_drain.self VERSION 01.00 NAME "VBE Gate-1C B-Stage Probe Drain")
vita_create_self(vbe_gate1c_status.self vbe_gate1c_status UNSAFE)
vita_create_vpk(vbe_gate1c_status.vpk VBEG1CST0 vbe_gate1c_status.self VERSION 01.00 NAME "VBE Gate-1C B-Stage Probe Status")

cmake_minimum_required(VERSION 3.10)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(DEFINED ENV{VITASDK})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain file")
  else()
    message(FATAL_ERROR "Please define VITASDK")
  endif()
endif()
project(vbe_gate1d_control C)
include("${VITASDK}/share/vita.cmake" REQUIRED)
if(NOT DEFINED VBE_PRODUCTION_STUB)
  message(FATAL_ERROR "Set VBE_PRODUCTION_STUB to libvitabright_stub.a")
endif()
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -Werror -O2 -std=gnu99")
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/../../..)

function(vbe_gate1d_app target source titleid name)
  add_executable(${target} ${source} common.c)
  target_link_libraries(${target} ${VBE_PRODUCTION_STUB} SceIofilemgr_stub SceLibKernel_stub)
  set(VITA_ELF_CREATE_FLAGS "")
  set(VITA_MAKE_FSELF_FLAGS "")
  vita_create_self(${target}.self ${target} UNSAFE)
  vita_create_vpk(${target}.vpk ${titleid} ${target}.self VERSION 01.00 NAME "${name}")
endfunction()

vbe_gate1d_app(vbe_gate1d_mild mild.c VBEG1DM90 "VBE Gate-1D Mild Matrix")
vbe_gate1d_app(vbe_gate1d_reset reset.c VBEG1DRS0 "VBE Gate-1D Matrix Reset")
vbe_gate1d_app(vbe_gate1d_status status.c VBEG1DST0 "VBE Gate-1D Matrix Status")

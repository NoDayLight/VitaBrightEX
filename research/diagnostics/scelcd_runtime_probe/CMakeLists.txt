cmake_minimum_required(VERSION 3.10)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(DEFINED ENV{VITASDK})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain file")
  else()
    message(FATAL_ERROR "Please define VITASDK")
  endif()
endif()
project(vbe_scelcd_runtime_probe C)
include("${VITASDK}/share/vita.cmake" REQUIRED)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -Werror -O2 -std=gnu99")
add_executable(vbe_scelcd_runtime_probe_kernel kernel.c)
target_link_libraries(vbe_scelcd_runtime_probe_kernel taihenForKernel_stub taihenModuleUtils_stub SceModulemgrForDriver_stub SceSysmemForDriver_stub gcc)
set_target_properties(vbe_scelcd_runtime_probe_kernel PROPERTIES LINK_FLAGS "-nostdlib" COMPILE_FLAGS "-D__VITA_KERNEL__ -fno-pic -fno-PIC")
vita_create_self(vbe_scelcd_runtime_probe.skprx vbe_scelcd_runtime_probe_kernel CONFIG ${CMAKE_CURRENT_SOURCE_DIR}/module.yml UNSAFE)
vita_create_stubs(vbe_scelcd_runtime_probe_stubs vbe_scelcd_runtime_probe_kernel ${CMAKE_CURRENT_SOURCE_DIR}/module.yml KERNEL)
add_executable(vbe_scelcd_runtime_probe_drain drain.c)
add_dependencies(vbe_scelcd_runtime_probe_drain vbe_scelcd_runtime_probe_stubs)
target_link_libraries(vbe_scelcd_runtime_probe_drain ${CMAKE_CURRENT_BINARY_DIR}/libVbeSceLcdRuntimeProbe_stub.a SceIofilemgr_stub)
set(VITA_ELF_CREATE_FLAGS "")
set(VITA_MAKE_FSELF_FLAGS "")
vita_create_self(vbe_scelcd_runtime_probe_drain.self vbe_scelcd_runtime_probe_drain UNSAFE)
vita_create_vpk(vbe_scelcd_runtime_probe.vpk VBESIG365 vbe_scelcd_runtime_probe_drain.self VERSION 01.00 NAME "VBE SceLcd C9 Probe")

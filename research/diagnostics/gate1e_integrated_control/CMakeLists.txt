cmake_minimum_required(VERSION 3.10)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
  if(DEFINED ENV{VITASDK})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VITASDK}/share/vita.toolchain.cmake" CACHE PATH "toolchain file")
  else()
    message(FATAL_ERROR "Please define VITASDK")
  endif()
endif()
project(vbe_gate1e_integrated_control C)
include("${VITASDK}/share/vita.cmake" REQUIRED)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -Werror -O2 -std=gnu99")
if(NOT VBE_PRODUCTION_STUB)
  message(FATAL_ERROR "VBE_PRODUCTION_STUB must point to the exact integrated libvitabright_stub.a")
endif()

function(add_gate1e_client target source titleid title)
  add_executable(${target} ${source} common.c)
  target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../..)
  target_link_libraries(${target} ${VBE_PRODUCTION_STUB} SceIofilemgr_stub SceLibKernel_stub)
  set(VITA_ELF_CREATE_FLAGS "")
  set(VITA_MAKE_FSELF_FLAGS "")
  vita_create_self(${target}.self ${target} UNSAFE)
  vita_create_vpk(${target}.vpk ${titleid} ${target}.self VERSION 01.00 NAME "${title}")
endfunction()

add_gate1e_client(vbe_gate1e_integrated_mild mild.c VBEG1EIM1 "VBE Gate-1E Integrated Mild")
add_gate1e_client(vbe_gate1e_integrated_reset reset.c VBEG1EIR1 "VBE Gate-1E Integrated Reset")
add_gate1e_client(vbe_gate1e_integrated_status status.c VBEG1EIS1 "VBE Gate-1E Integrated Status")
add_gate1e_client(vbe_gate1e_integrated_rollback rollback.c VBEG1ERB1 "VBE Gate-1E Rollback Test")
